\newcommand{\CJKnumber}[1]{%
  \if\relax\detokenize{#1}\relax\else\CJKnumeral{#1}\fi}

  \ifcase\numexpr#1\relax
    \or 一\or 二\or 三\or 四\or 五\or 六\or 七\or 八\or 九\or 十%
    \else #1\fi}

%%%%%%%%%%%% 封面 %%%%%%%%%%%%%%%%%
% 与原报告保持一致：直接使用原报告第一页的整页海报图作为封面。
% 若需换回天大封面（校名 + 校徽 + 学院/专业/年级/姓名/学号），
% 可用 git 历史恢复 figures/Tjulogo.eps、figures/tju.eps 并替换此处的 \makecover。
\newcommand{\makecover}{%
  \begin{titlepage}%
    \thispagestyle{empty}%
    \AddToShipoutPictureBG*{%
      \AtPageLowerLeft{\includegraphics[width=\paperwidth,height=\paperheight]{figures/cover-word.png}}}%
    \null
  \end{titlepage}%
  \ClearShipoutPictureBG
}
