%%%%%%%%%%%%%%%%% 中文数字 %%%%%%%%%%%%%%%%%
% 模板原先依赖 CJK 宏包的 \CJKnumber，迁移到 xeCJK 后自行实现（1–10 足够）。
\newcommand{\CJKnumber}[1]{%
  \if\relax\detokenize{#1}\relax\else\CJKnumeral{#1}\fi}
\newcommand{\CJKnumeral}[1]{%
  \ifcase\numexpr#1\relax
    \or 一\or 二\or 三\or 四\or 五\or 六\or 七\or 八\or 九\or 十%
    \else #1\fi}
% 目录中不编号章（如“证明材料”）的 \thecontentslabel 为空，此处仍提供标签命令备用。
\newcommand{\ChapterTocLabel}[1]{%
  \if\relax\detokenize{#1}\relax\else 第\CJKnumber{#1}章\quad\fi}
